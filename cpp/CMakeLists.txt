cmake_minimum_required(VERSION 3.16)
project(orchard_row_guidance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-O3 -Wall -Wextra)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui)
find_package(realsense2 QUIET)
add_library(orchard_guidance
  src/perception.cpp src/velocity.cpp src/guidance.cpp src/visualize.cpp src/synthetic.cpp)
target_include_directories(orchard_guidance PUBLIC include)
target_link_libraries(orchard_guidance PUBLIC ${OpenCV_LIBS})
find_package(Threads REQUIRED)
add_executable(tractor_guidance src/main.cpp src/preview_server.cpp)
target_link_libraries(tractor_guidance PRIVATE orchard_guidance Threads::Threads)
if(realsense2_FOUND)
  target_compile_definitions(tractor_guidance PRIVATE HAVE_REALSENSE=1)
  target_link_libraries(tractor_guidance PRIVATE realsense2::realsense2)
endif()
add_executable(test_guidance src/test_guidance.cpp)
target_link_libraries(test_guidance PRIVATE orchard_guidance)
